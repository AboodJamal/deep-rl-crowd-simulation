num_trials = base_env.num_trials
successes = 0

for trial_idx in range(num_trials):
    base_env.current_trial = trial_idx
    obs = vec_env.reset()
    
    done = False
    steps = 0
    while not done and steps < 500:
        action, _ = model.predict(obs, deterministic=True)
        obs, _, done, info = vec_env.step(action)
        steps += 1
    
    if info[0].get('goal_reached', False):
        successes += 1

print(f'MOSP_B: {successes}/{num_trials} ({100*successes/num_trials:.1f}%)')

obs, info = env.reset()
print(f'Initial observation shape: {obs.shape}')

total_reward = 0
steps = 0
while True:
    action = env.action_space.sample()  # Random action
    obs, reward, terminated, truncated, info = env.step(action)
    total_reward += reward
    steps += 1
    
    if terminated or truncated or steps > 500:
        break

print(f'Episode ended after {steps} steps')
print(f'Total reward: {total_reward:.2f}')
print(f'Goal reached: {info.get(\"goal_reached\", False)}')

================================================================================
IMPORTANT NOTES
================================================================================

1. VecNormalize is CRITICAL:
   - ALWAYS use --vec-normalize flag when evaluating
   - Without it, observations are not scaled correctly
   - The agent WILL fail without proper normalization

2. Deterministic vs Stochastic:
   - Use --deterministic for reproducible results
   - Remove flag for stochastic policy sampling

3. Model Files Required:
   - vga_drl_final.zip (the trained policy)
   - vga_drl_stage6_vecnormalize.pkl (observation normalizer)
   
4. Hardware:
   - GPU significantly speeds up training
   - CPU-only evaluation is fine

5. Data Path:
   - Modify data_root if VGA data is in different location
