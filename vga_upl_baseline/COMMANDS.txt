from models.vga_upl_planner_v4 import VGAUPLPlannerV4
import numpy as np

obstacles = [
    {'position': [3, 0], 'radius': 0.25},
    {'position': [6, 0.3], 'radius': 0.25},
    {'position': [8, -0.2], 'radius': 0.25}
]

paths = []
planner = VGAUPLPlannerV4(use_probabilistic=True)

for i in range(10):
    result = planner.simulate(
        start_pos=np.array([0, 0]),
        goal_pos=np.array([10, 0]),
        obstacles=obstacles
    )
    # Hash path by final y-coordinate sign at each obstacle
    path_sig = tuple(
        'L' if pos[1] > 0 else 'R' 
        for pos in result.positions[::50] 
        if 2 < pos[0] < 9
    )[:3]
    paths.append(path_sig)

from collections import Counter
print('Path distribution (10 runs):')
for path, count in Counter(paths).most_common():
    print(f'  {path}: {count} times ({count*10}%)')
"

# Example of using VGA+UPL programmatically
cd "C:\Users\Abdullah Jamal\Desktop\College\Forth year\First Term\training\bot3-enhanced\test-2\trains on everything- eval on everything\vga_upl_baseline"

# Import the planner
from models.vga_upl_planner_v4 import VGAUPLPlannerV4
import numpy as np

# Create planner (deterministic)
planner = VGAUPLPlannerV4(
    use_probabilistic=False,  # Set True for stochastic
    agent_radius=0.2,
    min_clearance=0.02,
    desired_speed=1.34
)

# Define scenario
start = np.array([0, 0])
goal = np.array([10, 0])
obstacles = [
    {'position': [3, 0], 'radius': 0.25},
    {'position': [5, 0.5], 'radius': 0.25},
    {'position': [5, -0.5], 'radius': 0.25},
    {'position': [7, 0], 'radius': 0.25},
]

# Run simulation
result = planner.simulate(start, goal, obstacles=obstacles, max_steps=2000)

# Access results
print(f'Success: {result.success}')
print(f'Positions shape: {result.positions.shape}')
print(f'Velocities shape: {result.velocities.shape}')
print(f'')
print('Metrics:')
for key, value in result.metadata.items():
    if isinstance(value, (int, float)):
        print(f'  {key}: {value:.4f}' if isinstance(value, float) else f'  {key}: {value}')
"

================================================================================
NOTES
================================================================================

1. Data Path:
   - If VGA data is in a different location, modify 'data_root' in scripts
   - Or edit experimental_data_config.yaml

2. Output Customization:
   - Edit NUM_VIDEOS, NUM_IMAGES in generate_v4_visualizations.py
   - Edit NUM_STOCHASTIC_RUNS in generate_stochastic_visualizations.py

3. Parameters:
   - agent_radius=0.2m (robot body)
   - min_clearance=0.02m (safety margin)
   - desired_speed=1.34m/s (Weidmann walking speed)
   - max_steps=2000 (timeout ~100 seconds)

4. Modes:
   - use_probabilistic=False → Deterministic (100% reproducible)
   - use_probabilistic=True → Stochastic (path variety)
